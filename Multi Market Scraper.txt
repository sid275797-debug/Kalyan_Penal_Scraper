import os
import json
import time
from datetime import timedelta
import pandas as pd
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By

# ===== CONFIG =====
OUTPUT_DIR = "output"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# Load markets JSON
with open("all_market_urls_cleaned.json", "r") as f:
    MARKETS = json.load(f)

# ===== Scraper & Cleaner =====
def scrape_market(market_name, url):
    options = Options()
    # options.add_argument("--headless")  # Uncomment for headless mode
    options.add_argument("--disable-gpu")
    options.add_argument("--no-sandbox")

    driver = webdriver.Chrome(options=options)
    print(f"\n🔍 Scraping {market_name}...")
    driver.get(url)
    time.sleep(8)  # Adjust as needed

    records = []
    try:
        table = driver.find_element(By.CSS_SELECTOR, "table.pchart")
        rows = table.find_elements(By.TAG_NAME, "tr")

        for row in rows:
            ths = row.find_elements(By.TAG_NAME, "th")
            tds = row.find_elements(By.TAG_NAME, "td")

            if len(ths) >= 3 and len(tds) >= 1:
                date_text = ths[0].get_attribute("innerHTML").replace("<br>", " ").strip()
                date = date_text.split("to")[0].strip()
                day = "Unknown"

                jodi_index = 0
                for i in range(1, len(ths) - 1, 2):
                    try:
                        open_panna = ths[i].text.strip().replace("\n", "")
                        close_panna = ths[i+1].text.strip().replace("\n", "")
                        jodi = tds[jodi_index].text.strip()

                        if len(jodi) == 2:
                            open_digit = jodi[0]
                            close_digit = jodi[1]

                            records.append({
                                "Date": date,
                                "Day": day,
                                "Open Panna": open_panna,
                                "Jodi": jodi,
                                "Close Panna": close_panna,
                                "Open Digit": open_digit,
                                "Close Digit": close_digit
                            })
                        jodi_index += 1
                    except:
                        continue

    except Exception as e:
        print(f"❌ Error scraping {market_name}: {e}")

    driver.quit()

    if records:
        df = pd.DataFrame(records)
        df = fix_dates(df)
        out_file = os.path.join(OUTPUT_DIR, f"{market_name}.csv")
        df.to_csv(out_file, index=False)
        print(f"✅ Saved: {out_file}")
    else:
        print(f"⚠️ No data for {market_name}")

# ===== Fix Dates & Days =====
def fix_dates(df):
    for i, row in df.iterrows():
        try:
            base_date = pd.to_datetime(row['Date'], dayfirst=True, errors='raise')
            start_index = i
            break
        except:
            continue
    else:
        return df

    for j in range(start_index, len(df)):
        new_date = base_date + timedelta(days=(j - start_index))
        df.at[j, 'Date'] = new_date.strftime('%d/%m/%Y')
        df.at[j, 'Day'] = new_date.strftime('%A')

    return df

# ===== MAIN =====
if __name__ == "__main__":
    for m in MARKETS:
        scrape_market(m['name'], m['url'])

    print("\n🎉 DONE: All markets attempted!")

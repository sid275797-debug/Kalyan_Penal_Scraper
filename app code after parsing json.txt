/// Models & Services
class UnlockCode {
  final String code;
  final String tier;
  final DateTime expires;
  UnlockCode({required this.code, required this.tier, required this.expires});
  factory UnlockCode.fromJson(Map<String, dynamic> j) {
    return UnlockCode(code: (j['code'] as String).trim(), tier: j['tier'] as String, expires: DateTime.parse(j['expires'] as String));
  }
}

class Prediction {
  final String marketKey;
  final Map<String, List<String>> p;
  double score = 0.0;
  bool passed = false;
  Prediction({required this.marketKey, required this.p});
  factory Prediction.fromJson(Map<String, dynamic> j) {
    final key = (j['market'] as String).toLowerCase();
    final raw = j['predictions'] as Map<String, dynamic>;
    final map = <String, List<String>>{};
    raw.forEach((k, v) {
      map[k] = (v as List).map((x) => x.toString()).toList();
    });
    return Prediction(marketKey: key, p: map);
  }
}

class Yesterday {
  final String marketKey;
  final String openDigit, closeDigit, openPanna, closePanna, jodi;
  Yesterday({required this.marketKey, required this.openDigit, required this.closeDigit, required this.openPanna, required this.closePanna, required this.jodi});
  factory Yesterday.fromJson(Map<String, dynamic> j) {
    final key = (j['market'] as String).toLowerCase();
    return Yesterday(
      marketKey: key,
      openDigit: j['Open Digit'].toString(),
      closeDigit: j['Close Digit'].toString(),
      openPanna: j['Open Panna'].toString(),
      closePanna: j['Close Panna'].toString(),
      jodi: j['Jodi'].toString(),
    );
  }
}

class ScorePredictor {
  final double threshold;
  ScorePredictor({this.threshold = 0.6});
  List<Prediction> annotateAndFilter(List<Prediction> preds) {
    for (final p in preds) {
      final double s = computeScoreFromPrediction(p);
      p.score = double.parse(s.toStringAsFixed(3));
      p.passed = s >= threshold;
    }
    return preds.where((p) => p.passed).toList();
  }

  double computeScoreFromPrediction(Prediction p) {
    double? confidence;
    try {
      if (p.p.containsKey('confidence') && p.p['confidence']!.isNotEmpty) {
        confidence = double.tryParse(p.p['confidence']!.first);
      }
    } catch (_) { confidence = null; }

    double? recent;
    try {
      if (p.p.containsKey('recent_success_rate') && p.p['recent_success_rate']!.isNotEmpty) {
        recent = double.tryParse(p.p['recent_success_rate']!.first);
      }
    } catch (_) { recent = null; }

    const double modelWeight = 0.45;
    const double jodiWeight = 0.12;
    const double openPannaWeight = 0.08;
    const double closePannaWeight = 0.08;
    const double digitsWeight = 0.12;
    const double premiumWeight = 0.05;
    const double hasDateWeight = 0.05;
    const double recentWeight = 0.05;

    double heuristicsScore = 0.0;
    final hasJodi = (p.p['jodis'] != null && p.p['jodis']!.isNotEmpty && p.p['jodis']!.first.trim().isNotEmpty);
    heuristicsScore += (hasJodi ? 1.0 : 0.0) * jodiWeight;

    final openPannas = p.p['open_pannas'] ?? <String>[];
    final double openNorm = (openPannas.length >= 5) ? 1.0 : (openPannas.length / 5.0);
    heuristicsScore += openNorm * openPannaWeight;

    final closePannas = p.p['close_pannas'] ?? <String>[];
    final double closeNorm = (closePannas.length >= 5) ? 1.0 : (closePannas.length / 5.0);
    heuristicsScore += closeNorm * closePannaWeight;

    final hasOpenDigit = (p.p['open_digits'] != null && p.p['open_digits']!.isNotEmpty && p.p['open_digits']!.first.trim().isNotEmpty);
    final hasCloseDigit = (p.p['close_digits'] != null && p.p['close_digits']!.isNotEmpty && p.p['close_digits']!.first.trim().isNotEmpty);
    heuristicsScore += ((hasOpenDigit && hasCloseDigit) ? 1.0 : 0.0) * digitsWeight;

    final nameUpper = p.marketKey.toUpperCase();
    final isPremium = nameUpper.contains('KALYAN') || nameUpper.contains('RAJDHANI') || nameUpper.contains('MILAN');
    heuristicsScore += (isPremium ? 1.0 : 0.0) * premiumWeight;

    final hasDate = (p.p['date'] != null && p.p['date']!.isNotEmpty && p.p['date']!.first.trim().isNotEmpty);
    heuristicsScore += (hasDate ? 1.0 : 0.0) * hasDateWeight;

    if (recent != null && recent >= 0 && recent <= 1) heuristicsScore += recent * recentWeight;

    if (confidence != null && confidence >= 0 && confidence <= 1) {
      final double finalScore = (confidence * modelWeight) + heuristicsScore;
      return finalScore.clamp(0.0, 1.0);
    } else {
      const double heuristicSumWeight = 1.0 - modelWeight;
      final double normalized = (heuristicSumWeight > 0) ? (heuristicsScore / heuristicSumWeight) : 0.0;
      return normalized.clamp(0.0, 1.0);
    }
  }
}

class PredictionService {
  static const _remoteUrl = 'https://pranitborkar98.github.io/sattaboss-data/data/todays_predictions.json';
  static Future<List<Prediction>> load() async {
    try {
      final resp = await http.get(Uri.parse(_remoteUrl));
      if (resp.statusCode == 200) {
        final decoded = json.decode(resp.body);
        final list = (decoded is List)
            ? decoded
            : (decoded is Map && decoded['data'] is List)
            ? decoded['data']
            : [];        return list.map((e) => Prediction.fromJson(e)).toList();
      }
    } catch (_) {}
    final raw = await rootBundle.loadString('assets/todays_predictions.json');
    final list = json.decode(raw) as List<dynamic>;
    return list.map((e) => Prediction.fromJson(e)).toList();
  }
}

class YesterdayService {
  static const _remoteUrl = 'https://pranitborkar98.github.io/sattaboss-data/data/yesterday_results.json';
  static Future<List<Yesterday>> load() async {

    final list = json.decode(raw) as List<dynamic>;
    return list.map((e) => Yesterday.fromJson(e)).toList();
  }
}

class _HomeScreenState extends State<HomeScreen> with SingleTickerProviderStateMixin {
  late final TabController _tabController;
  bool _loading = true;

  // domain
  List<Prediction> _allPreds = [];
  List<Yesterday> _allYest = [];
  late Map<String, Prediction> _predByKey;
  late Map<String, Yesterday> _histByKey;
  late List<Prediction> _freePreds;
  late List<Prediction> _premiumPreds;

  // gating
  final Set<int> _tabsUnlocked = {0};

  // score
  List<Prediction> _scorePassed = [];
  late ScorePredictor _scorePredictor;

  List<ScoreMarket> _scoreMarkets = [];
  bool _scoreAssetLoaded = false;

  // sangam
  List<SangamMarket> _sangamData = [];
  bool _sangamsGenerated = false;

  // subscription
  bool _isPremiumUnlocked = false;
  DateTime? _subscriptionExpiry;
  final Set<String> _usedCodes = {};
  List<UnlockCode> _codes = [];
  final TextEditingController _codeController = TextEditingController();

  // ads
  late BannerAd _bannerAd;
  bool _isBannerAdReady = false;
  RewardedAd? _rewardedAd;
  bool _isRewardedAdReady = false;

  final _premiumPatterns = ['kalyan', 'rajdhani', 'milan', 'main_bazar', 'main_bombay', 'time_bazar', 'supreme'];

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: 6, vsync: this);
    _scorePredictor = ScorePredictor(threshold: 0.6);
    _loadAllData();
    _loadCodesAndSubscription();
    _loadBannerAd();
    _loadRewardedAd();
    _initFirebaseMessaging();

    _tabController.addListener(() {
      if (!_tabController.indexIsChanging) return;
      final targetIndex = _tabController.index;
      if (targetIndex == 0) return;
      if (!_tabsUnlocked.contains(targetIndex)) {
        final prev = _tabController.previousIndex;
        _tabController.index = prev;
        _showRewardedAd(() {
          _tabsUnlocked.add(targetIndex);
          _tabController.animateTo(targetIndex);
        });
      }
    });
  }

  void _initFirebaseMessaging() {
    FirebaseMessaging.onMessage.listen((RemoteMessage message) {
      final notif = message.notification;
      if (notif != null && notif.title != null) {
        _showSnack('${notif.title}: ${notif.body ?? ''}');
      }
    });
    FirebaseMessaging.onMessageOpenedApp.listen((RemoteMessage message) {
      final notif = message.notification;
      if (notif != null && notif.title != null) {
        _showSnack("Tapped: ${notif.title}");
      }
    });
  }

  @override
  void dispose() {
    _tabController.dispose();
    _codeController.dispose();
    _bannerAd.dispose();
    _rewardedAd?.dispose();
    super.dispose();
  }

  Future<void> _loadAllData() async {
    List<Prediction> preds = [];
    List<Yesterday> yest = [];
    List<UnlockCode> unlockCodes = [];

    Future<String?> _loadAsset(String path) async {
      try {
        return await rootBundle.loadString(path);
      } catch (e) {
        debugPrint('ASSET MISSING: $path -> $e');
        _showSnack('Missing asset: $path');
        return null;
      }
    }

    dynamic _safeJsonDecode(String src, String label) {
      try {
        return json.decode(src);
      } catch (e) {
        debugPrint('JSON DECODE FAILED ($label): $e\nFirst 120: ${src.length > 120 ? src.substring(0, 120) : src}');
        _showSnack('Corrupt $label JSON');
        return null;
      }
    }

    try { preds = await PredictionService.load(); } catch (e) { debugPrint('PredictionService error: $e'); }
    try { yest = await YesterdayService.load(); } catch (e) { debugPrint('YesterdayService error: $e'); }

    final rawCodes = await _loadAsset('assets/unlock_codes.json');
    if (rawCodes != null) {
      final decoded = _safeJsonDecode(rawCodes, 'unlock_codes');
      if (decoded is List) {
        try {
          unlockCodes = decoded.whereType<Map<String, dynamic>>().map((e) => UnlockCode.fromJson(e)).toList();
        } catch (e) {
          debugPrint('UnlockCode bulk parse failed: $e. Falling back per-entry.');
          unlockCodes = [];
          for (final row in decoded) {
            try { if (row is Map<String, dynamic>) unlockCodes.add(UnlockCode.fromJson(row)); } catch (e) { debugPrint('Skipping bad unlock code row: $e  row=$row'); }
          }
        }
      } else {
        debugPrint('unlock_codes.json is not a JSON array; ignoring.');
      }
    }

    final predMap = {for (var p in preds) p.marketKey: p};
    final histMap = {for (var h in yest) h.marketKey: h};
    final prem = preds.where((p) => _premiumPatterns.any((pat) => p.marketKey.contains(pat))).toList();
    final free = preds.where((p) => !prem.contains(p)).toList();
    final passed = _scorePredictor.annotateAndFilter(preds);

    setState(() {
      _allPreds = preds;
      _allYest = yest;
      _predByKey = predMap;
      _histByKey = histMap;
      _premiumPreds = prem;
      _freePreds = free;
      _codes = unlockCodes;
      _scorePassed = passed;
      _loading = false;
    });

    await _loadScoreAsset();
    await _loadSangamsFromPrefs();
  }

  Future<void> _loadCodesAndSubscription() async {
    final prefs = await SharedPreferences.getInstance();
    final used = prefs.getStringList('usedCodes') ?? [];
    final expString = prefs.getString('subscriptionExpiry');
    DateTime? exp;
    if (expString != null) {
      exp = DateTime.tryParse(expString);
      if (exp != null && exp.isAfter(DateTime.now())) {
        _isPremiumUnlocked = true;
        _subscriptionExpiry = exp;
      } else {
        await prefs.remove('usedCodes');
        await prefs.remove('subscriptionExpiry');
      }
    }
    setState(() => _usedCodes.addAll(used));
  }

  void _loadBannerAd() {
    _bannerAd = BannerAd(
      adUnitId: 'ca-app-pub-3940256099942544/6300978111',
      request: const AdRequest(),
      size: AdSize.banner,
      listener: BannerAdListener(
        onAdLoaded: (_) => setState(() => _isBannerAdReady = true),
        onAdFailedToLoad: (ad, err) {
          ad.dispose();
          debugPrint('BannerAd failed to load: $err');
        },
      ),
    )..load();
  }

  void _loadRewardedAd() {
    RewardedAd.load(
      adUnitId: 'ca-app-pub-3940256099942544/5224354917',
      request: const AdRequest(),
      rewardedAdLoadCallback: RewardedAdLoadCallback(
        onAdLoaded: (ad) {
          _rewardedAd = ad;
          _isRewardedAdReady = true;
        },
        onAdFailedToLoad: (err) {
          debugPrint('RewardedAd failed: $err');
          _isRewardedAdReady = false;
        },
      ),
    );
  }

  void _showRewardedAd(VoidCallback onCompletion) {
    if (_isRewardedAdReady && _rewardedAd != null) {
      _rewardedAd!.fullScreenContentCallback = FullScreenContentCallback(
        onAdDismissedFullScreenContent: (ad) {
          ad.dispose();
          _loadRewardedAd();
          onCompletion();
        },
        onAdFailedToShowFullScreenContent: (ad, err) {
          ad.dispose();
          _loadRewardedAd();
          onCompletion();
        },
      );
      // show expects a callback with two params; using (ad, reward)
      _rewardedAd!.show(onUserEarnedReward: (ad, reward) {});
    } else {
      onCompletion();
    }
  }

  Future<void> _launch(String url) async {
    final uri = Uri.parse(url);
    if (await canLaunchUrl(uri)) {
      await launchUrl(uri, mode: LaunchMode.externalApplication);
    }
  }

  Future<void> _unlockWithCode(String code) async {
    final rawInput = code;
    code = code.trim().toUpperCase();

    if (_codes.isEmpty) {
      debugPrint('UNLOCK: No codes loaded. Check asset bundling.');
      _showSnack('No unlock codes loaded. Rebuild with assets.');
      return;
    }

    final found = _codes.where((u) => u.code == code).toList();
    if (found.isEmpty) {
      debugPrint('UNLOCK: Code not found. input="$rawInput" normalized="$code" loadedCodes=${_codes.length} exampleFirst="${_codes.first.code}"');
      _showSnack('Code not found');
      return;
    }

    final match = found.first;
    final now = DateTime.now();
    if (!match.expires.isAfter(now)) {
      debugPrint('UNLOCK: Code found but expired. code=$code now=$now expires=${match.expires}');
      _showSnack('Code expired on ${match.expires.toLocal().toShortDateString()}');
      return;
    }

    if (_usedCodes.contains(code)) {
      debugPrint('UNLOCK: Code already used. code=$code');
      _showSnack('Already used');
      return;
    }

    final prefs = await SharedPreferences.getInstance();
    _usedCodes.add(code);
    await prefs.setStringList('usedCodes', _usedCodes.toList());
    await prefs.setString('subscriptionExpiry', match.expires.toIso8601String());

    setState(() {
      _isPremiumUnlocked = true;
      _subscriptionExpiry = match.expires;
    });

    debugPrint('UNLOCK: Success. code=$code expires=${match.expires}');
    _showSnack('Unlocked until ${match.expires.toLocal().toShortDateString()}');
    _tabController.animateTo(0);
  }

  String _metricLabel(String metric) {
    switch (metric) {
      case 'open_digit': return 'Open';
      case 'close_digit': return 'Close';
      case 'open_panna': return 'Open Panna';
      case 'close_panna': return 'Close Panna';
      case 'jodi': return 'Jodi';
      default:
        return metric.splitMapJoin(RegExp(r'_'), onMatch: (_) => ' ', onNonMatch: (s) => s).trim().split(' ').map((w) => w.isEmpty ? w : (w[0].toUpperCase() + w.substring(1))).join(' ');
    }

  String _uiDisplayFor(String metric, dynamic value) {
    if (value == null) return '';
    if (metric == 'open_digit' || metric == 'close_digit') return '↑ ${value.toString()} — PASS';
    if (metric == 'open_panna' || metric == 'close_panna') return '→ ${value.toString()} — PASS';
    if (metric == 'jodi') return '★ ${value.toString()} — PASS';
    return value.toString();
  }

  Future<void> _loadScoreAsset() async {
    try {
      // Try fetching live score JSON from GitHub first
      final response = await http.get(Uri.parse(
        'https://raw.githubusercontent.com/pranitborkar98/sattaboss-data/main/sattaboss-data/data/prediction_score_report.json',
      ));

      Map<String, dynamic> doc = {};
      if (response.statusCode == 200) {
        final decoded = json.decode(response.body);
        if (decoded is Map<String, dynamic>) {
          doc = decoded;
        } else {
          // unexpected shape — fallback to local asset
          final raw = await rootBundle.loadString('assets/prediction_score_report.json');
          final localDecoded = json.decode(raw);
          if (localDecoded is Map<String, dynamic>) doc = localDecoded;
        }
      } else {
        // non-200 — fallback to bundled asset
        final raw = await rootBundle.loadString('assets/prediction_score_report.json');
        final localDecoded = json.decode(raw);
        if (localDecoded is Map<String, dynamic>) doc = localDecoded;
      }

      final Map<String, dynamic> marketsMap = (doc['markets'] as Map<String, dynamic>?) ?? {};
      final List<ScoreMarket> parsed = [];
      marketsMap.forEach((key, entry) {
        try {
          if (entry is Map<String, dynamic>) {
            final hits = entry['hits'] as Map<String, dynamic>? ?? {};
            final actual = entry['actual'] as Map<String, dynamic>? ?? {};
            final Map<String, ScoreHit> hitsOnly = {};
            hits.forEach((metric, val) {
              final passed = (val == true);
              if (passed) {
                final dynVal = actual.containsKey(metric) ? actual[metric] : actual[_mapMetricToActualKey(metric)];
                final display = _uiDisplayFor(metric, dynVal);
                hitsOnly[metric] = ScoreHit(metric: metric, value: dynVal, display: display);
              }
            });
            if (hitsOnly.isNotEmpty) {
              final isPremium = _premiumPatterns.any((pat) => key.contains(pat));
              final displayName = _display(key);
              parsed.add(ScoreMarket(market: displayName, key: key, premium: isPremium, hits: hitsOnly));
            }
          }
        } catch (e) {
          debugPrint('Score asset parse skipped market=$key -> $e');
        }
      });

      parsed.sort((a, b) {
        if (a.premium && !b.premium) return -1;
        if (!a.premium && b.premium) return 1;
        return a.market.compareTo(b.market);
      });

      setState(() {
        _scoreMarkets = parsed;
        _scoreAssetLoaded = true;
      });
    } catch (e) {
      debugPrint('Score asset load failed: $e');
      setState(() {
        _scoreMarkets = [];
        _scoreAssetLoaded = true;
      });
    }
    }
  String _mapMetricToActualKey(String metric) => metric;

  // ---------------- Sangam generation (3 per market) ----------------
  List<SangamMarket> _generateSangamsForAllMarkets() {
    final List<SangamMarket> result = [];
    final rng = Random();
    if (_allPreds.isEmpty) return result;

    final all = List<Prediction>.from(_allPreds);
    all.sort((a, b) {
      final aPrem = _premiumPatterns.any((pat) => a.marketKey.contains(pat));
      final bPrem = _premiumPatterns.any((pat) => b.marketKey.contains(pat));
      if (aPrem && !bPrem) return -1;
      if (!aPrem && bPrem) return 1;
      return a.marketKey.compareTo(b.marketKey);
    });

    for (final p in all) {
      final openPannas = p.p['open_pannas'] ?? <String>[];
      final closePannas = p.p['close_pannas'] ?? <String>[];
      final openDigits = p.p['open_digits'] ?? <String>[];
      final closeDigits = p.p['close_digits'] ?? <String>[];
      if (openPannas.isEmpty || closePannas.isEmpty || openDigits.isEmpty || closeDigits.isEmpty) continue;

      final Set<String> sangams = {};
      // attempt up to 12 draws but keep only up to 3 unique sangams
      for (int i = 0; i < 12 && sangams.length < 3; i++) {
        final oP = openPannas[rng.nextInt(openPannas.length)];
        final cP = closePannas[rng.nextInt(closePannas.length)];
        final oD = openDigits[rng.nextInt(openDigits.length)];
        final cD = closeDigits[rng.nextInt(closeDigits.length)];
        // format consistent with examples like "123-65-140" you mentioned — using pattern used before
        final s = '${oP.toString()}-${oD.toString()}${cD.toString()}-${cP.toString()}';
        sangams.add(s);
      }

      if (sangams.isNotEmpty) {
        final isPrem = _premiumPatterns.any((pat) => p.marketKey.contains(pat));
        result.add(SangamMarket(market: _display(p.marketKey), premium: isPrem, sangams: sangams.toList()));
      }
    }

    return result;
  }

  Future<void> _saveSangamsToPrefs() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final now = DateTime.now();
      final expiry = now.add(const Duration(hours: 12)).toIso8601String();
      final List<Map<String, dynamic>> serial = _sangamData.map((m) => {'market': m.market, 'premium': m.premium, 'sangams': m.sangams}).toList();
      await prefs.setString('sangam_data_json', jsonEncode(serial));
      await prefs.setString('sangam_data_expiry', expiry);
      debugPrint('Saved sangams -> expiry=$expiry count=${serial.length}');
    } catch (e) {
      debugPrint('Failed to save sangams: $e');
    }
  }

  Future<void> _loadSangamsFromPrefs() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final raw = prefs.getString('sangam_data_json');
      final expiryStr = prefs.getString('sangam_data_expiry');
      if (raw == null || expiryStr == null) return;
      final expiry = DateTime.tryParse(expiryStr);
      if (expiry == null || expiry.isBefore(DateTime.now())) {
        await prefs.remove('sangam_data_json');
        await prefs.remove('sangam_data_expiry');
        debugPrint('Saved sangams expired — cleared');
        return;
      }
      final List<dynamic> decoded = json.decode(raw) as List<dynamic>;
      final List<SangamMarket> loaded = decoded.map((e) {
        final mm = e as Map<String, dynamic>;
        return SangamMarket(market: mm['market'] as String, premium: mm['premium'] as bool, sangams: (mm['sangams'] as List<dynamic>).map((x) => x.toString()).toList());
      }).toList();
      setState(() {
        _sangamData = loaded;
        _sangamsGenerated = _sangamData.isNotEmpty;
      });
      debugPrint('Loaded saved sangams (${_sangamData.length}) — valid until $expiryStr');
    } catch (e) {
      debugPrint('Failed to load saved sangams: $e');
    }
  }

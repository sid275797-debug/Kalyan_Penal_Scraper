String _prettyMarketName(String raw) {
  var s = (raw ?? '').toString();
  s = s.replaceAll(RegExp(r'[_\.]+'), ' ').trim();
  final parts = s.split(' ').where((w) => w.isNotEmpty).toList();
  return parts.map((w) => w[0].toUpperCase() + w.substring(1)).join(' ');
}
// --- Compatibility helpers for Prediction shapes -----------------------
// These helpers accept either a `Prediction` instance (if your model class
// exposes fields) or a plain Map (Map<String,dynamic>) coming from RTDB.
// They use `dynamic` to avoid compile-time brittle access and provide a
// normalized Map<String,dynamic> representation we can safely mutate/read.

Map<String, dynamic> _predictionToMap(dynamic p) {
  // If it's already a Map, copy and return
  try {
    if (p is Map<String, dynamic>) return Map<String, dynamic>.from(p);
    if (p is Map) return Map<String, dynamic>.from(p.map((k, v) => MapEntry(k.toString(), v)));
  } catch (_) {}

  // If it's an object (Prediction class), try dynamic access (toJson(), fields)
  try {
    final dyn = p as dynamic;
    // prefer .toJson() if available
    try {
      final j = dyn.toJson();
      if (j is Map) return Map<String, dynamic>.from(j.map((k, v) => MapEntry(k.toString(), v)));
    } catch (_) {}
    // fallback: try common property names
    final market = (dyn.market ?? dyn.marketKey ?? dyn.name) ?? '';
    final payload = (dyn.p ?? dyn.predictions ?? dyn.data) ?? <String, dynamic>{};
    final score = (dyn.score ?? 0.0);
    final passed = (dyn.passed ?? false);
    return {
      'market': market,
      'p': payload is Map ? Map<String, dynamic>.from(payload.map((k, v) => MapEntry(k.toString(), v))) : payload,
      'score': score,
      'passed': passed,
    };
  } catch (_) {}

  // ultimate fallback
  return <String, dynamic>{'market': '', 'p': <String, dynamic>{}, 'score': 0.0, 'passed': false};
}

/// Create a Prediction instance from a dynamic/map if your Prediction has
/// a fromJson factory. If not, this attempts to call Prediction.fromJson
/// and otherwise just returns the input (best-effort).
dynamic _mapToPrediction(Map<String, dynamic> m) {
  try {
    // try to call the constructor via dynamic (works if fromJson exists)
    final ctor = (Prediction as dynamic).fromJson;
    if (ctor != null) {
      return (Prediction as dynamic).fromJson(m);
    }
  } catch (_) {}
  // if no fromJson or something else, return the raw map (app code might handle it)
  return m;
}

/// Safely get the inner predictions map (the field often named `p`)
Map<String, dynamic> _getInnerP(dynamic pObj) {
  try {
    final dyn = pObj as dynamic;
    // if pObj is Prediction instance
    if (pObj is! Map) {
      final candidate = (dyn.p ?? dyn.predictions ?? dyn.data);
      if (candidate is Map) return Map<String, dynamic>.from(candidate.map((k, v) => MapEntry(k.toString(), v)));
      if (candidate is List) return {'list': candidate};
    } else {
      // if it's already a map
      final m = (pObj as Map).map((k, v) => MapEntry(k.toString(), v));
      if (m.containsKey('p')) {
        final inner = m['p'];
        if (inner is Map) return Map<String, dynamic>.from(inner.map((k, v) => MapEntry(k.toString(), v)));
      }
      // maybe already contains keys like "Open Digits" at top-level
      return Map<String, dynamic>.from(m);
    }
  } catch (_) {}
  return <String, dynamic>{};
}

/// Helper to coerce any list-like field into List<String>
List<String> toStrList(dynamic v) {
  if (v == null) return <String>[];
  if (v is List) return v.map((e) => e?.toString() ?? '').where((s) => s.isNotEmpty).toList();
  if (v is String) return <String>[v];
  return <String>[];
}

class ScorePredictor {
  final double threshold;
  ScorePredictor({this.threshold = 0.2});

  /// Annotate predictions immutably (returning a list of updated Prediction objects)
  /// and then return only those that pass the threshold.
  List<Prediction> annotateAndFilter(List<Prediction> preds) {
    // create annotated (copied) list with score & passed fields set via copyWith
    final annotated = preds.map((p) {
      final double s = computeScoreFromPrediction(p);
      final bool passed = s >= threshold;


// old: return p.copyWith(score: s, passed: passed);

// new:
      final Map<String, dynamic> _pmap = _predictionToMap(p);
      _pmap['score'] = s;
      _pmap['passed'] = passed;
      final dynamic newPred = _mapToPrediction(_pmap);
      return newPred;    }).toList();

    print('[ScorePredictor] annotated count=${annotated.length}');



    // keep a separate list containing annotated items that passed
// old: return annotated.where((p) => p.passed).toList();

// new:
    final filtered = annotated.where((item) {
      try {
        final dyn = item as dynamic;
        final val = (dyn.passed ?? (item is Map ? item['passed'] : null));
        return val == true;
      } catch (_) {
        return false;
      }
    }).toList();
    print('[ScorePredictor] filtered (passed) count=${filtered.length}');


// Ensure result is List<Prediction> if possible
    final List<Prediction> outList = <Prediction>[];
    for (final item in filtered) {
      if (item is Prediction) {
        outList.add(item);
      } else if (item is Map) {
        final p = _mapToPrediction(Map<String, dynamic>.from(item));
        if (p is Prediction) outList.add(p);
      }
    }

    print('[ScorePredictor] returning ${outList.length} items to UI');

// If we couldn't map to Prediction, but we still need to return something of that type,
// returning the map list will likely cause a static type mismatch — but usually
// your UI expects either Prediction or compatible map. If your code requires
// strict typing, change function signature to return List<dynamic>.
    return outList;  }




  double computeScoreFromPrediction(Prediction p) {
    // Helper to coerce many shapes into a List<dynamic>
    List<dynamic> toList(dynamic v) {
      if (v == null) return <dynamic>[];
      if (v is List) return v;
      // sometimes value might be a String like "2025-10-15" or a single number
      return <dynamic>[v];
    }

    // Use safe lookup; p.p is now an alias to predictions (map)
    final Map<String, dynamic> m = _getInnerP(p);
    final openDigits = toList(m['open_digits']).map((e) => e?.toString()).whereType<String>().toList();
    final closeDigits = toList(m['close_digits']).map((e) => e?.toString()).whereType<String>().toList();
    final openPannas = toList(m['open_pannas']).map((e) => e?.toString()).whereType<String>().toList();
    final closePannas = toList(m['close_pannas']).map((e) => e?.toString()).whereType<String>().toList();
    final jodis = toList(m['jodis']).map((e) => e?.toString()).whereType<String>().toList();

    // date may be a list or a single string — pick first safely
    final dateList = toList(m['date']);
    final dateStr = dateList.isNotEmpty ? dateList.first.toString() : '';

    // Example scoring logic (keeps your previous logic intent).
    // Change weighting according to your original algorithm.
    double score = 0.0;
    // + some points for non-empty fields
    if (openDigits.isNotEmpty) score += 0.25;
    if (closeDigits.isNotEmpty) score += 0.20;
    if (openPannas.isNotEmpty) score += 0.20;
    if (closePannas.isNotEmpty) score += 0.20;
    if (jodis.isNotEmpty) score += 0.10;

    // tiny boost if date exists
    if (dateStr.isNotEmpty) score += 0.01;

    // clamp to <= 1.0
    if (score > 1.0) score = 1.0;
    return score;
  }
}

  Future<void> _loadAllData() async {
    List<Prediction> preds = [];
    List<Yesterday> yest = [];
    List<UnlockCode> unlockCodes = [];

    Future<String?> loadAsset(String path) async {
      try {
        return await rootBundle.loadString(path);
      } catch (e) {
        debugPrint('ASSET MISSING: $path -> $e');
        _showSnack('Missing asset: $path');
        return null;
      }
    }

    dynamic safeJsonDecode(String src, String label) {
      try {
        return json.decode(src);
      } catch (e) {
        debugPrint('JSON DECODE FAILED ($label): $e\nFirst 120: ${src.length > 120 ? src.substring(0, 120) : src}');
        _showSnack('Corrupt $label JSON');
        return null;
      }
    }

    try { preds = await PredictionService().load(); } catch (e) { debugPrint('PredictionService error: $e'); }
    try {
      final score = await _scoreService.load();
      setState(() {
        scoreData = score;
        _scoresLoading = false;
      });
    } catch (e) {
      debugPrint('ScoreService load failed: $e');
      setState(() {
        scoreError = e.toString();
        _scoresLoading = false;
      });
    }

    final rawCodes = await loadAsset('assets/unlock_codes.json');
    if (rawCodes != null) {
      final decoded = safeJsonDecode(rawCodes, 'unlock_codes');
      if (decoded is List) {
        try {
          unlockCodes = decoded.whereType<Map<String, dynamic>>().map((e) => UnlockCode.fromJson(e)).toList();
        } catch (e) {
          debugPrint('UnlockCode bulk parse failed: $e. Falling back per-entry.');
          unlockCodes = [];
          for (final row in decoded) {
            try { if (row is Map<String, dynamic>) unlockCodes.add(UnlockCode.fromJson(row)); } catch (e) { debugPrint('Skipping bad unlock code row: $e  row=$row'); }
          }
        }
      } else {
        debugPrint('unlock_codes.json is not a JSON array; ignoring.');
      }

        final Map<String, dynamic> actual =

                  // sometimes digits are keyed without "_digit"

  Future<void> _unlockWithCode(String inputCode) async {
    if (inputCode.trim().isEmpty) {
      _showSnack('Please enter a code');
      return;
    }

    final code = inputCode.replaceAll(RegExp(r'\s+'), '').toUpperCase();
    debugPrint('🔓 Trying unlock code: $code');

    if (_codes.isEmpty) {
      _showSnack('Unlock codes not loaded yet');
      debugPrint('⚠️ No unlock codes loaded');
      return;
    }

    final found = _codes.where((u) {
      try {
        final stored = ((u as dynamic).code ?? '').toString().replaceAll(RegExp(r'\s+'), '').toUpperCase();
        return stored == code;
      } catch (e) {
        return false;
      }
    }).toList();

    if (found.isEmpty) {
      _showSnack('Code not found');
      debugPrint('❌ Code not found: $code');
      return;
    }

    final unlock = found.first;

    // expiry check
    try {
      final dyn = unlock as dynamic;
      final expiresVal = dyn.expires ?? dyn.expiry ?? dyn.expiresAt;
      if (expiresVal != null) {
        final expiry = DateTime.tryParse(expiresVal.toString());
        if (expiry != null && expiry.isBefore(DateTime.now())) {
          _showSnack('This code has expired');
          debugPrint('⚠️ Expired code: ${dyn.code}');
          return;
        }
      }
    } catch (e) {
      debugPrint('Expiry check error: $e');
      // continue — don't block unlock for parse failure
    }

    // used-check (try common properties)
    try {
      final dyn = unlock as dynamic;
      var isUsed = false;
      try {
        if ((dyn.used ?? false) == true) isUsed = true;
      } catch (_) {}
      try {
        if ((dyn.isUsed ?? false) == true) isUsed = true;
      } catch (_) {}
      try {
        if ((dyn.usedAt ?? '').toString().isNotEmpty) isUsed = true;
      } catch (_) {}
      if (isUsed || _usedCodes.contains(code)) {
        _showSnack('This code has already been used');
        debugPrint('⚠️ Code already used: ${dyn.code}');
        return;
      }
    } catch (e) {
      debugPrint('Used-check error (non-fatal): $e');
    }

    // --- APPLY UNLOCK: mark expiry, persist and update UI state ---
    try {
      DateTime unlockUntil;
      try {
        final dyn = (unlock as dynamic);
        final expiresVal = dyn.expires ?? dyn.expiry ?? dyn.expiresAt;
        if (expiresVal != null) {
          final parsed = DateTime.tryParse(expiresVal.toString());
          unlockUntil = parsed ?? DateTime.now().add(const Duration(days: 365));
        } else {
          unlockUntil = DateTime.now().add(const Duration(days: 365));
        }
      } catch (_) {
        unlockUntil = DateTime.now().add(const Duration(days: 365));
      }

      // Persist expiry and usedCodes
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString('subscription_expiry', unlockUntil.toIso8601String()); // consistent key
      // add code to usedCodes list and persist
      _usedCodes.add(code);
      await prefs.setStringList('usedCodes', _usedCodes.toList());

      // mark as used on the unlock entry if possible (non-fatal)
      try {
        final dyn = unlock as dynamic;
        try { dyn.used = true; } catch (_) {}
        try { dyn.isUsed = true; } catch (_) {}
        try { dyn.usedAt = DateTime.now().toIso8601String(); } catch (_) {}
      } catch (_) {}

      // Update runtime flags and UI
      setState(() {
        _subscriptionExpiry = unlockUntil;
        _isPremiumUnlocked = true;
      });

      final formattedDate = '${unlockUntil.year}-${unlockUntil.month.toString().padLeft(2, '0')}-${unlockUntil.day.toString().padLeft(2, '0')}';
      _showSnack('Premium unlocked until $formattedDate');
      debugPrint('✅ Unlock success: ${(unlock as dynamic).code ?? "unknown"} → $formattedDate');

    } catch (e) {
      debugPrint('❌ Unlock apply failed: $e');
      _showSnack('Unlock failed, try again');
    }
  }

  String _display(String? key) {

        .where((w) => w.isNotEmpty)
        .map((w) => w[0].toUpperCase() + (w.length > 1 ? w.substring(1) : ''))
        .join(' ');
  }

  void _listenYesterdayRTDB() {
    final ref = FirebaseDatabase.instance.ref('sattaboss/yesterday_results');

    ref.onValue.listen((event) {
      if (!event.snapshot.exists) {
        debugPrint('⚠️ No RTDB data for Yesterday Winners');
        return;
      }

      final Map allDates = Map<String, dynamic>.from(event.snapshot.value as Map);
      if (allDates.isEmpty) return;

      // pick latest date node
      final dates = allDates.keys.map((e) => e.toString()).toList()..sort();
      final latest = dates.last;

      final Map markets = Map<String, dynamic>.from(allDates[latest] as Map);

      final list = markets.entries.map((e) {
        final d = Map<String, dynamic>.from(e.value as Map);
        return Yesterday(
          marketKey: e.key,
          openDigit: d['Open Digit']?.toString() ?? '',
          closeDigit: d['Close Digit']?.toString() ?? '',
          openPanna: d['Open Panna']?.toString() ?? '',
          closePanna: d['Close Panna']?.toString() ?? '',
          jodi: d['Jodi']?.toString() ?? '',
        );
      }).toList();

      setState(() {
        _allYest = list;
        _histByKey = { for (final h in list) h.marketKey : h };
      });

      debugPrint('✅ RTDB: Loaded ${list.length} Yesterday markets for $latest');
    });
  }

// Filter only markets with at least one passed hit or score_details.pass == true

        bool hasTrue = false;

          v is Map &&

// ---------------- Sangam generation (3 per market) ----------------
  List<SangamMarket> _generateSangamsForAllMarkets() {
    final List<SangamMarket> result = [];
    if (_allPreds.isEmpty) return result;

    // -----------------------------
    // FIXED PANNA MAP (LOCAL - your finalized lists)
    // -----------------------------
    final Map<int, List<String>> fixedPannasByDigit = {
      0: ["118","127","136","145","190","226","235","244","280","334","370","460","550","299","389","479","488","569","578","668","677"],
      1: ["100","119","128","137","146","155","227","236","245","290","335","344","380","470","560","399","489","579","588","669","678"],
      2: ["110","200","129","138","147","156","228","237","246","255","336","345","390","480","570","660","499","589","679","688","778"],
      3: ["777","444","120","300","111","139","148","157","166","229","238","247","256","337","346","355","445","490","580","670","599","689","779","788"],
      4: ["112","130","220","400","149","158","167","239","248","257","266","338","347","356","446","455","590","680","770","699","789"],
      5: ["113","122","140","230","500","159","168","177","249","258","267","339","348","357","366","447","456","690","780","799","889"],
      6: ["888","555","114","123","150","240","330","600","222","169","178","259","268","277","349","358","367","448","457","466","556","790","880","899"],
      7: ["115","124","133","160","223","250","340","700","179","188","269","278","359","368","377","449","458","467","557","566","890"],
      8: ["116","125","134","170","224","233","260","350","440","800","189","279","288","369","378","459","468","477","558","567","990"],
      9: ["199","289","379","388","469","478","559","568","577","667","999","666","117","126","135","144","180","225","234","270","360","450","900","333"],
    };
    // -----------------------------
    // End of fixed panna map
    // -----------------------------

    // helper lists
    final List<int> digitsPresent = fixedPannasByDigit.keys.toList();
    final List<String> allFixedPannas = fixedPannasByDigit.values.expand((l) => l).toSet().toList();

    // deterministic date seed (same day => same sangams)
    final dateSeed = DateTime.now().toIso8601String().split('T').first;

    // preserve your sort: premium markets first
    final all = List<Prediction>.from(_allPreds);
    all.sort((a, b) {
      final aPrem = _premiumPatterns.any((pat) => a.marketKey.contains(pat));
      final bPrem = _premiumPatterns.any((pat) => b.marketKey.contains(pat));
      if (aPrem && !bPrem) return -1;
      if (!aPrem && bPrem) return 1;
      return a.marketKey.compareTo(b.marketKey);
    });

    // safe digit parser: extract an int 0..9 fallback to first available fixed digit
    int parseDigitSafe(String d) {
      final s = d.trim();
      if (s.isEmpty) return digitsPresent.isNotEmpty ? digitsPresent.first : 0;
      try {
        final val = int.parse(s);
        if (val >= 0 && val <= 9) return val;
        // if multi-digit (like '10'), take last digit
        return int.parse(s[s.length - 1]);
      } catch (_) {
        // try to extract last numeric char
        final m = RegExp(r'\d').allMatches(s).map((e) => s[e.start]).toList();
        if (m.isNotEmpty) {
          try { return int.parse(m.last); } catch (_) {}
        }
        return digitsPresent.isNotEmpty ? digitsPresent.first : 0;
      }
    }

    // pick panna from fixed list for digit, safe fallback to any fixed panna
    String pickFromFixed(int digit, Random rng) {
      final list = fixedPannasByDigit[digit] ?? [];
      if (list.isNotEmpty) return list[rng.nextInt(list.length)];
      // fallback
      return allFixedPannas.isNotEmpty ? allFixedPannas[rng.nextInt(allFixedPannas.length)] : '000';
    }

    for (final p in all) {
      // defensive extraction
      final dynamic openPannasRaw = _getInnerP(p)['open_pannas'] ?? _getInnerP(p)['Open Pannas'];

      final dynamic closePannasRaw = _getInnerP(p)['close_pannas'] ?? _getInnerP(p)['Close Pannas'];

      final dynamic openDigitsRaw = _getInnerP(p)['open_digits'] ?? _getInnerP(p)['Open digits'];

      final dynamic closeDigitsRaw = _getInnerP(p)['close_digits'] ?? _getInnerP(p)['close digits'];

      final openPannas = <String>[];
      final closePannas = <String>[];
      final openDigits = <String>[];
      final closeDigits = <String>[];

      if (openPannasRaw != null && openPannasRaw is Iterable) {
        for (final dynamic x in openPannasRaw) {
          final s = x?.toString() ?? '';
          if (s.trim().isNotEmpty) openPannas.add(s);
        }
      }
      if (closePannasRaw != null && closePannasRaw is Iterable) {
        for (final dynamic x in closePannasRaw) {
          final s = x?.toString() ?? '';
          if (s.trim().isNotEmpty) closePannas.add(s);
        }
      }
      if (openDigitsRaw != null && openDigitsRaw is Iterable) {
        for (final dynamic x in openDigitsRaw) {
          final s = x?.toString() ?? '';
          if (s.trim().isNotEmpty) openDigits.add(s);
        }
      }
      if (closeDigitsRaw != null && closeDigitsRaw is Iterable) {
        for (final dynamic x in closeDigitsRaw) {
          final s = x?.toString() ?? '';
          if (s.trim().isNotEmpty) closeDigits.add(s);
        }
      }

      // If digits missing, skip
      if (openDigits.isEmpty || closeDigits.isEmpty) continue;

      // seeded RNG per market/day for deterministic output
      final seedString = '${p.marketKey}::$dateSeed';
      final int folded = seedString.runes.fold<int>(0, (int prev, int r) => prev * 31 + r);
      final int seed = folded & 0x7fffffff;
      final rng = Random(seed);

      final Set<String> sangams = {};
      final attemptsLimit = 60; // try to build 3 unique sangams

      for (int attempt = 0; attempt < attemptsLimit && sangams.length < 3; attempt++) {
        final oDStr = openDigits[rng.nextInt(openDigits.length)];
        final cDStr = closeDigits[rng.nextInt(closeDigits.length)];
        final oD = parseDigitSafe(oDStr);
        final cD = parseDigitSafe(cDStr);

        // PICK FROM FIXED lists only (no mismatch)
        final oP = pickFromFixed(oD, rng);
        final cP = pickFromFixed(cD, rng);

        // final formatted sangam, e.g. '140-57-270' (openP - openDigitcloseDigit - closeP)
        final sang = '$oP-${oD.toString()}${cD.toString()}-$cP';
        sangams.add(sang);
      }

      if (sangams.isNotEmpty) {
        final isPrem = _premiumPatterns.any((pat) => p.marketKey.contains(pat));
        result.add(SangamMarket(market: _display(p.marketKey), premium: isPrem, sangams: sangams.toList().take(3).toList()));
      }
    }

    return result;

  Future<void> _loadSangamsFromPrefs() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final raw = prefs.getString('sangam_data_json');
      final expiryStr = prefs.getString('sangam_data_expiry');
      if (raw == null || expiryStr == null) return;
      final expiry = DateTime.tryParse(expiryStr);
      if (expiry == null || expiry.isBefore(DateTime.now())) {
        await prefs.remove('sangam_data_json');
        await prefs.remove('sangam_data_expiry');
        debugPrint('Saved sangams expired — cleared');
        return;
      }
      final List<dynamic> decoded = json.decode(raw) as List<dynamic>;
      final List<SangamMarket> loaded = decoded.map((e) {
        final mm = e as Map<String, dynamic>;
        return SangamMarket(market: mm['market'] as String, premium: mm['premium'] as bool, sangams: (mm['sangams'] as List<dynamic>).map((x) => x.toString()).toList());
      }).toList();
      setState(() {
        _sangamData = loaded;
        _sangamsGenerated = _sangamData.isNotEmpty;
      });
      debugPrint('Loaded saved sangams (${_sangamData.length}) — valid until $expiryStr');
    } catch (e) {
      debugPrint('Failed to load saved sangams: $e');
    }
  }
